// Command server serves the calculator API (and optionally the static frontend).
package main

import (
        "context"
        "errors"
        "flag"
        "log/slog"
        "net/http"
        "os"
        "os/signal"
        "syscall"
        "time"

        "calculator/internal/api"
)

func main() {
        healthcheck := flag.Bool("healthcheck", false, "GET /healthz on localhost and exit 0 if healthy")
        flag.Parse()

        port := os.Getenv("PORT")
        if port == "" {
                port = "8080"
        }

        if *healthcheck {
                os.Exit(check(port))
        }

        logger := slog.New(slog.NewJSONHandler(os.Stdout, nil))
        if err := run(logger, ":"+port, os.Getenv("STATIC_DIR")); err != nil {
                logger.Error("server failed", "error", err)
                os.Exit(1)
        }
}

func run(logger *slog.Logger, addr, staticDir string) error {
        srv := &http.Server{
                Addr:              addr,
                Handler:           api.NewRouter(logger, staticDir),
                ReadHeaderTimeout: 2 * time.Second,
                ReadTimeout:       5 * time.Second,
                WriteTimeout:      10 * time.Second,
                IdleTimeout:       60 * time.Second,
        }

        ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt, syscall.SIGTERM)
        defer stop()

        errc := make(chan error, 1)
        go func() { errc <- srv.ListenAndServe() }()
        logger.Info("listening", "addr", addr)

        select {
        case err := <-errc:
                return err
        case <-ctx.Done():
        }

        logger.Info("shutting down")
        shutdownCtx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := srv.Shutdown(shutdownCtx); err != nil && !errors.Is(err, http.ErrServerClosed) {
                return err
        }
        return nil
}

// check returns the process exit code: 0 if /healthz answers 200, else 1.
func check(port string) int {
        client := &http.Client{Timeout: 2 * time.Second}
        resp, err := client.Get("http://127.0.0.1:" + port + "/healthz")
        if err != nil {
                return 1
        }
        defer resp.Body.Close()
        if resp.StatusCode != http.StatusOK {
                return 1
        }
        return 0
}

// API-level codes, not produced by the calc package.
const (
        CodeInvalidJSON      calc.Code = "INVALID_JSON"
        CodePayloadTooLarge  calc.Code = "PAYLOAD_TOO_LARGE"
        codeNotFound         calc.Code = "NOT_FOUND"
        codeMethodNotAllowed calc.Code = "METHOD_NOT_ALLOWED"
)

var statusFor = map[calc.Code]int{
        calc.CodeInvalidExpression: 400,
        calc.CodeExpressionTooLong: 400,
        calc.CodeInvalidOperands:   400,
        CodeInvalidJSON:            400,
        calc.CodeUnknownOperation:  404,
        calc.CodeDivisionByZero:    422,
        calc.CodeInvalidOperation:  422,
        calc.CodeOverflow:          422,
        CodePayloadTooLarge:        413,
}

// status maps a code to its HTTP status, defaulting to 500 so WriteHeader never gets 0.
func status(code calc.Code) int {
        if s, ok := statusFor[code]; ok {
                return s
        }
        return 500
}

type errBody struct {
        Code     calc.Code `json:"code"`
        Message  string    `json:"message"`
        Position *int      `json:"position,omitempty"`
}

func writeJSON(w http.ResponseWriter, status int, v any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        json.NewEncoder(w).Encode(v)
}

func writeErr(w http.ResponseWriter, status int, b errBody) {
        writeJSON(w, status, map[string]errBody{"error": b})
}

func writeCodeErr(w http.ResponseWriter, code calc.Code, msg string) {
        writeErr(w, status(code), errBody{Code: code, Message: msg})
}

func writeCalcErr(w http.ResponseWriter, err error) {
        var ce *calc.Error
        if !errors.As(err, &ce) {
                writeErr(w, 500, errBody{Code: "INTERNAL", Message: "Internal error"})
                return
        }
        b := errBody{Code: ce.Code, Message: ce.Message}
        if ce.Position >= 0 {
                b.Position = &ce.Position
        }
        writeErr(w, status(ce.Code), b)
}

func notFound(w http.ResponseWriter, _ *http.Request) {
        writeErr(w, 404, errBody{Code: codeNotFound, Message: "Not found"})
}

func methodNotAllowed(w http.ResponseWriter, _ *http.Request) {
        writeErr(w, 405, errBody{Code: codeMethodNotAllowed, Message: "Method not allowed"})
}

const maxBody = 4096

// decode reads a strict JSON body into v; on failure it writes the error response and returns false.
func decode(w http.ResponseWriter, r *http.Request, v any) bool {
        dec := json.NewDecoder(http.MaxBytesReader(w, r.Body, maxBody))
        dec.DisallowUnknownFields()
        if err := dec.Decode(v); err != nil {
                var tooBig *http.MaxBytesError
                if errors.As(err, &tooBig) {
                        writeCodeErr(w, CodePayloadTooLarge, "Invalid input")
                } else {
                        writeCodeErr(w, CodeInvalidJSON, "Invalid input")
                }
                return false
        }
        return true
}

func evaluate(w http.ResponseWriter, r *http.Request) {
        var req struct {
                Expression string `json:"expression"`
        }
        if !decode(w, r, &req) {
                return
        }
        res, err := calc.Evaluate(req.Expression)
        if err != nil {
                writeCalcErr(w, err)
                return
        }
        out := struct {
                Result float64 `json:"result"`
                Repeat string  `json:"repeat,omitempty"`
        }{res.Value, res.Repeat}
        writeJSON(w, 200, out)
}

func operation(w http.ResponseWriter, r *http.Request) {
        var req struct {
                Operands []float64 `json:"operands"`
        }
        if !decode(w, r, &req) {
                return
        }
        v, err := calc.Apply(chi.URLParam(r, "name"), req.Operands...)
        if err != nil {
                writeCalcErr(w, err)
                return
        }
        writeJSON(w, 200, map[string]float64{"result": calc.Round(v)})
}

func healthz(w http.ResponseWriter, _ *http.Request) {
        writeJSON(w, 200, map[string]string{"status": "ok"})
}

import (

        "time"

        "github.com/go-chi/chi/v5"

)

// NewRouter builds the HTTP handler. staticDir "" disables static/SPA serving.
func NewRouter(logger *slog.Logger, staticDir string) http.Handler {
        r := chi.NewRouter()
        r.Use(middleware.Recoverer, requestLogger(logger))

        r.NotFound(notFound)
        if staticDir != "" {
                r.NotFound(spaHandler(staticDir))
        }
        r.MethodNotAllowed(methodNotAllowed)

        r.Get("/healthz", healthz)
        r.Route("/api/v1", func(r chi.Router) {
                r.Post("/evaluate", evaluate)
                r.Post("/operations/{name}", operation)
        })
        return r
}

func requestLogger(logger *slog.Logger) func(http.Handler) http.Handler {
        return func(next http.Handler) http.Handler {
                return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                        start := time.Now()
                        ww := middleware.NewWrapResponseWriter(w, r.ProtoMajor)
                        next.ServeHTTP(ww, r)
                        logger.Info("request",
                                "method", r.Method, "path", r.URL.Path, "status", ww.Status(),
                                "duration_ms", time.Since(start).Milliseconds())
                })
        }
}

// spaHandler serves an existing file under dir, else index.html. /api/ paths always get the JSON 404.
func spaHandler(dir string) http.HandlerFunc {
        return func(w http.ResponseWriter, r *http.Request) {
                if strings.HasPrefix(r.URL.Path, "/api/") {
                        notFound(w, r)
                        return
                }
                f := filepath.Join(dir, filepath.FromSlash(path.Clean("/"+r.URL.Path)))
                if st, err := os.Stat(f); err != nil || st.IsDir() {
                        f = filepath.Join(dir, "index.html")
                }
                http.ServeFile(w, r, f)
        }
}

type Code string

const (
        CodeInvalidExpression Code = "INVALID_EXPRESSION"
        CodeExpressionTooLong Code = "EXPRESSION_TOO_LONG"
        CodeInvalidOperands   Code = "INVALID_OPERANDS"
        CodeUnknownOperation  Code = "UNKNOWN_OPERATION"
        CodeDivisionByZero    Code = "DIVISION_BY_ZERO"
        CodeInvalidOperation  Code = "INVALID_OPERATION"
        CodeOverflow          Code = "OVERFLOW"
)

// Error is a typed calculation error. Position is -1 when not applicable.
type Error struct {
        Code     Code
        Message  string
        Position int
}

func (e *Error) Error() string { return e.Message }

var messages = map[Code]string{
        CodeDivisionByZero:   "Undefined",
        CodeOverflow:         "Overflow",
        CodeUnknownOperation: "Unknown operation",
}

func newError(code Code, pos int) *Error {
        msg, ok := messages[code]
        if !ok {
                msg = "Invalid input"
        }
        return &Error{Code: code, Message: msg, Position: pos}
}

package calc

import (
        "errors"
        "strconv"
        "strings"
)

type TokenKind int

const (
        TokNumber TokenKind = iota
        TokPlus
        TokMinus
        TokStar
        TokSlash
        TokCaret
        TokPercent
        TokSqrt
        TokLParen
        TokRParen
        TokEOF
)

// Token.Pos is the byte offset of the token's first character.
type Token struct {
        Kind  TokenKind
        Value float64
        Pos   int
}

var punct = map[byte]TokenKind{
        '+': TokPlus, '-': TokMinus, '*': TokStar, '/': TokSlash,
        '^': TokCaret, '%': TokPercent, '(': TokLParen, ')': TokRParen,
}

func isDigit(c byte) bool { return c >= '0' && c <= '9' }

// Lex tokenizes input; the result always ends with TokEOF.
func Lex(input string) ([]Token, error) {
        var toks []Token
        for i := 0; i < len(input); {
                c := input[i]
                switch {
                case c == ' ' || c == '\t' || c == '\n' || c == '\r':
                        i++
                case isDigit(c) || c == '.':
                        start := i
                        for i < len(input) && isDigit(input[i]) {
                                i++
                        }
                        if i < len(input) && input[i] == '.' {
                                i++
                                fracStart := i
                                for i < len(input) && isDigit(input[i]) {
                                        i++
                                }
                                if i == fracStart && start == fracStart-1 { // lone "."
                                        return nil, newError(CodeInvalidExpression, start)
                                }
                        }
                        if i < len(input) && (input[i] == 'e' || input[i] == 'E') {
                                i++
                                if i < len(input) && (input[i] == '+' || input[i] == '-') {
                                        i++
                                }
                                expStart := i
                                for i < len(input) && isDigit(input[i]) {
                                        i++
                                }
                                if i == expStart {
                                        return nil, newError(CodeInvalidExpression, start)
                                }
                        }
                        v, err := strconv.ParseFloat(input[start:i], 64)
                        if err != nil && !errors.Is(err, strconv.ErrRange) {
                                return nil, newError(CodeInvalidExpression, start)
                        }
                        toks = append(toks, Token{TokNumber, v, start})
                case strings.HasPrefix(input[i:], "sqrt"):
                        toks = append(toks, Token{Kind: TokSqrt, Pos: i})
                        i += 4
                default:
                        k, ok := punct[c]
                        if !ok {
                                return nil, newError(CodeInvalidExpression, i)
                        }
                        toks = append(toks, Token{Kind: k, Pos: i})
                        i++
                }
        }
        return append(toks, Token{Kind: TokEOF, Pos: len(input)}), nil
}

package calc

import (
        "math"
        "strconv"
)

type Operation struct {
        Arity int
        Fn    func(args []float64) (float64, error)
}

var Operations = map[string]Operation{
        "add":      {2, func(a []float64) (float64, error) { return a[0] + a[1], nil }},
        "subtract": {2, func(a []float64) (float64, error) { return a[0] - a[1], nil }},
        "multiply": {2, func(a []float64) (float64, error) { return a[0] * a[1], nil }},
        "divide": {2, func(a []float64) (float64, error) {
                if a[1] == 0 {
                        return 0, newError(CodeDivisionByZero, -1)
                }
                return a[0] / a[1], nil
        }},
        "power": {2, func(a []float64) (float64, error) {
                if a[0] == 0 && a[1] < 0 {
                        return 0, newError(CodeDivisionByZero, -1)
                }
                return math.Pow(a[0], a[1]), nil
        }},
        "sqrt":    {1, func(a []float64) (float64, error) { return math.Sqrt(a[0]), nil }},
        "percent": {1, func(a []float64) (float64, error) { return a[0] / 100, nil }},
}

// Apply runs an operation without rounding.
func Apply(name string, args ...float64) (float64, error) {
        op, ok := Operations[name]
        if !ok {
                return 0, newError(CodeUnknownOperation, -1)
        }
        if len(args) != op.Arity {
                return 0, newError(CodeInvalidOperands, -1)
        }
        r, err := op.Fn(args)
        if err != nil {
                return 0, err
        }
        if math.IsInf(r, 0) {
                return 0, newError(CodeOverflow, -1)
        }
        if math.IsNaN(r) {
                return 0, newError(CodeInvalidOperation, -1)
        }
        return r, nil
}

// Round rounds to 15 significant digits and normalizes -0 to 0.
func Round(x float64) float64 {
        r, _ := strconv.ParseFloat(strconv.FormatFloat(x, 'g', 15, 64), 64)
        if r == 0 {
                return 0
        }
        return r
}

package calc

import (
        "math"
        "strings"
)

const MaxExpressionLength = 256

type Result struct {
        Value  float64
        Repeat string
}

// operand is a parsed value; pct/raw mark a bare `n%` so expr can apply the contextual rule.
type operand struct {
        v, raw float64
        pct    bool
}

type parser struct {
        toks     []Token
        i        int
        depth    int
        repOp    int // position of the rightmost top-level binary operator
        repEnd   int // position just past its right operand
        hasRepOp bool
}

func (p *parser) peek() Token { return p.toks[p.i] }

func (p *parser) next() Token {
        t := p.toks[p.i]
        p.i++
        return t
}

func (p *parser) fail() error { return newError(CodeInvalidExpression, p.peek().Pos) }

// record notes a top-level binary operator; the largest opPos wins because
// lower-precedence operators are recorded after higher ones.
func (p *parser) record(opPos int) {
        if p.depth == 0 && (!p.hasRepOp || opPos > p.repOp) {
                p.hasRepOp, p.repOp, p.repEnd = true, opPos, p.peek().Pos
        }
}

func (p *parser) expr() (operand, error) {
        left, err := p.term()
        if err != nil {
                return left, err
        }
        for k := p.peek().Kind; k == TokPlus || k == TokMinus; k = p.peek().Kind {
                op := p.next()
                right, err := p.term()
                if err != nil {
                        return left, err
                }
                if right.pct {
                        if right.v, err = Apply("multiply", left.v, right.raw); err == nil {
                                right.v, err = Apply("percent", right.v)
                        }
                        if err != nil {
                                return left, err
                        }
                }
                p.record(op.Pos)
                name := "add"
                if k == TokMinus {
                        name = "subtract"
                }
                if left.v, err = Apply(name, left.v, right.v); err != nil {
                        return left, err
                }
                left.pct = false
        }
        return left, nil
}

func (p *parser) term() (operand, error) {
        left, err := p.unary()
        if err != nil {
                return left, err
        }
        for k := p.peek().Kind; k == TokStar || k == TokSlash; k = p.peek().Kind {
                op := p.next()
                right, err := p.unary()
                if err != nil {
                        return left, err
                }
                p.record(op.Pos)
                name := "multiply"
                if k == TokSlash {
                        name = "divide"
                }
                if left.v, err = Apply(name, left.v, right.v); err != nil {
                        return left, err
                }
                left.pct = false
        }
        return left, nil
}

func (p *parser) unary() (operand, error) {
        if p.peek().Kind != TokMinus {
                return p.power()
        }
        p.next()
        o, err := p.unary()
        return operand{v: -o.v}, err
}

func (p *parser) power() (operand, error) {
        base, err := p.prefix()
        if err != nil || p.peek().Kind != TokCaret {
                return base, err
        }
        op := p.next()
        exp, err := p.unary()
        if err != nil {
                return base, err
        }
        p.record(op.Pos)
        v, err := Apply("power", base.v, exp.v)
        return operand{v: v}, err
}

func (p *parser) prefix() (operand, error) {
        if p.peek().Kind != TokSqrt {
                return p.postfix()
        }
        p.next()
        o, err := p.prefix()
        if err != nil {
                return o, err
        }
        v, err := Apply("sqrt", o.v)
        return operand{v: v}, err
}

func (p *parser) postfix() (operand, error) {
        o, err := p.primary()
        if err != nil || p.peek().Kind != TokPercent {
                return o, err
        }
        p.next()
        v, err := Apply("percent", o.v)
        return operand{v: v, raw: o.v, pct: true}, err
}

func (p *parser) primary() (operand, error) {
        switch t := p.peek(); t.Kind {
        case TokNumber:
                p.next()
                return operand{v: t.Value}, nil
        case TokLParen:
                p.next()
                p.depth++
                o, err := p.expr()
                p.depth--
                if err != nil {
                        return o, err
                }
                if p.peek().Kind != TokRParen {
                        return o, p.fail()
                }
                p.next()
                return operand{v: o.v}, nil
        }
        return operand{}, p.fail()
}

// Evaluate parses and evaluates expr, returning the rounded value and the
// repeat fragment (last top-level binary operator plus its right operand).
func Evaluate(expr string) (Result, error) {
        if len(expr) > MaxExpressionLength {
                return Result{}, newError(CodeExpressionTooLong, -1)
        }
        toks, err := Lex(expr)
        if err != nil {
                return Result{}, err
        }
        p := &parser{toks: toks}
        o, err := p.expr()
        if err == nil && p.peek().Kind != TokEOF {
                err = p.fail()
        }
        if err != nil {
                return Result{}, err
        }
        // A bare literal such as 1e999 never passes through Apply.
        if math.IsInf(o.v, 0) {
                return Result{}, newError(CodeOverflow, -1)
        }
        if math.IsNaN(o.v) {
                return Result{}, newError(CodeInvalidOperation, -1)
        }
        res := Result{Value: Round(o.v)}
        if p.hasRepOp {
                res.Repeat = expr[p.repOp:p.repOp+1] + " " + strings.TrimSpace(expr[p.repOp+1:p.repEnd])
        }
        return res, nil
}
